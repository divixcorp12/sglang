cmake_minimum_required(VERSION 3.24)
project(nvfp4_cpu_bench LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The benchmark requires Linux worker-affinity verification")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED COMPONENTS CXX)
option(NVFP4_FETCH_BENCHMARK "Fetch pinned Google Benchmark if unavailable" ON)
find_package(benchmark 1.9 QUIET CONFIG)
if(NOT benchmark_FOUND)
  if(NOT NVFP4_FETCH_BENCHMARK)
    message(FATAL_ERROR "Install Google Benchmark or enable NVFP4_FETCH_BENCHMARK")
  endif()
  include(FetchContent)
  set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
  set(BENCHMARK_ENABLE_WERROR OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(google_benchmark
    GIT_REPOSITORY https://github.com/google/benchmark.git
    GIT_TAG eddb0241389718a23a42db6af5f0164b6e0139af) # same v1.9.4 as expert_stream/bench
  FetchContent_MakeAvailable(google_benchmark)
endif()
# The library's flags (optimized/build.py ARCH_FLAGS): baseline x86-64, the ISA tier picked at run time
# (NVFP4_CPU_MAX_ISA caps it), so the benchmark times the code production runs.
set(NVFP4_ARCH_FLAGS -march=x86-64 -mtune=generic)
add_library(nvfp4_kernel OBJECT ../optimized/kernel.cpp ../upstream/nvfp4.c)
target_link_libraries(nvfp4_kernel PRIVATE OpenMP::OpenMP_CXX)
target_compile_options(nvfp4_kernel PRIVATE -O3 -g -ffp-contract=off ${NVFP4_ARCH_FLAGS})
add_executable(nvfp4_cpu_optimized src/cpu_forward.cpp $<TARGET_OBJECTS:nvfp4_kernel>)
target_include_directories(nvfp4_cpu_optimized PRIVATE ../optimized)
target_compile_definitions(nvfp4_cpu_optimized PRIVATE NVFP4_BENCH_BACKEND="optimized")
target_compile_options(nvfp4_cpu_optimized PRIVATE -O3 -g -ffp-contract=off ${NVFP4_ARCH_FLAGS} -Wall -Wextra)
target_link_libraries(nvfp4_cpu_optimized PRIVATE benchmark::benchmark Threads::Threads OpenMP::OpenMP_CXX)
set_target_properties(nvfp4_cpu_optimized PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
