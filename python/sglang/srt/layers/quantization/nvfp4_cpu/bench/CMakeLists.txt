cmake_minimum_required(VERSION 3.24)
project(nvfp4_cpu_bench LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The benchmark requires Linux worker-affinity verification")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED COMPONENTS CXX)
option(NVFP4_FETCH_BENCHMARK "Fetch pinned Google Benchmark if unavailable" ON)
find_package(benchmark 1.9 QUIET CONFIG)
if(NOT benchmark_FOUND)
  if(NOT NVFP4_FETCH_BENCHMARK)
    message(FATAL_ERROR "Install Google Benchmark or enable NVFP4_FETCH_BENCHMARK")
  endif()
  include(FetchContent)
  set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
  set(BENCHMARK_ENABLE_WERROR OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(google_benchmark
    GIT_REPOSITORY https://github.com/google/benchmark.git
    GIT_TAG eddb0241389718a23a42db6af5f0164b6e0139af) # same v1.9.4 as expert_stream/bench
  FetchContent_MakeAvailable(google_benchmark)
endif()
foreach(backend baseline optimized)
  add_library(nvfp4_kernel_${backend} OBJECT ../optimized/moe_mul1.cpp ../upstream/nvfp4.c)
  target_link_libraries(nvfp4_kernel_${backend} PRIVATE OpenMP::OpenMP_CXX)
  target_compile_options(nvfp4_kernel_${backend} PRIVATE -O3 -g -ffp-contract=off -march=native)
  if(backend STREQUAL "baseline")
    target_compile_definitions(nvfp4_kernel_${backend} PRIVATE NVFP4_CPU_UPSTREAM_BASELINE=1)
  endif()
  add_executable(nvfp4_cpu_${backend} src/cpu_forward.cpp $<TARGET_OBJECTS:nvfp4_kernel_${backend}>)
  target_include_directories(nvfp4_cpu_${backend} PRIVATE ../optimized)
  target_compile_definitions(nvfp4_cpu_${backend} PRIVATE NVFP4_BENCH_BACKEND="${backend}")
  target_compile_options(nvfp4_cpu_${backend} PRIVATE -O3 -g -ffp-contract=off -march=native -Wall -Wextra)
  target_link_libraries(nvfp4_cpu_${backend} PRIVATE benchmark::benchmark Threads::Threads OpenMP::OpenMP_CXX)
  set_target_properties(nvfp4_cpu_${backend} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endforeach()
