cmake_minimum_required(VERSION 3.24)
project(sglang_nvfp4_cpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
option(NVFP4_CPU_NATIVE "Compile the plugin for this machine's CPU ISA" OFF)
option(NVFP4_BUILD_BENCHMARK "Build the native Google Benchmark executables" OFF)
add_library(sglang_nvfp4_cpu SHARED optimized/moe_mul1.cpp)
target_include_directories(sglang_nvfp4_cpu PUBLIC optimized)
target_compile_options(sglang_nvfp4_cpu PRIVATE -O3 -ffp-contract=off)
if(NVFP4_CPU_NATIVE)
  target_compile_options(sglang_nvfp4_cpu PRIVATE -march=native)
endif()
target_link_libraries(sglang_nvfp4_cpu PRIVATE Threads::Threads)
include(CTest)
if(BUILD_TESTING)
  get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
  add_executable(nvfp4_cpu_native_check "${REPO}/test/registered/unit/kernels/nvfp4_cpu_sanitizer.cpp")
  target_link_libraries(nvfp4_cpu_native_check PRIVATE sglang_nvfp4_cpu)
  # The standalone harness uses assert for hard correctness gates in Release.
  target_compile_options(nvfp4_cpu_native_check PRIVATE -UNDEBUG)
  add_test(NAME nvfp4_cpu_native_check COMMAND nvfp4_cpu_native_check)
endif()
if(NVFP4_BUILD_BENCHMARK)
  add_subdirectory(bench)
endif()
